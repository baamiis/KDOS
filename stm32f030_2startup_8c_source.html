37 * RCC and FLASH registers used during clock init
38 * ========================================================================= */
39#define FLASH_ACR (*(volatile uint32_t *)0x40022000UL)
40#define RCC_CR (*(volatile uint32_t *)0x40021000UL)
41#define RCC_CFGR (*(volatile uint32_t *)0x40021004UL)
42
43static void clock_init_48MHz(void)
44{
45 /* 1 NVM wait state required for 24–48 MHz; enable prefetch */
46 FLASH_ACR = (1u << 4) | 1u; /* PRFTBE | LATENCY=1 */
47
48 /* PLL: source = HSI/2 (PLLSRC=0, default), PLLMUL = ×12 (bits[21:18]=1010) */
49 RCC_CFGR = (10u << 18); /* PLLMUL = ×12 → 4 MHz × 12 = 48 MHz */
50
51 /* Enable PLL, wait ready */
52 RCC_CR |= (1u << 24); /* PLLON */
53 while (!(RCC_CR & (1u << 25))) {} /* PLLRDY */
54
55 /* Switch system clock to PLL, wait for switch */
56 RCC_CFGR |= 2u; /* SW = PLL */
57 while ((RCC_CFGR & (3u << 2)) != (2u << 2)) {} /* SWS = PLL */
58}

152/* =========================================================================
153 * Reset handler
154 * ========================================================================= */
155void Reset_Handler(void)
156{
157 clock_init_48MHz();
158
159 /* Copy .data from flash to RAM */
160 const uint32_t *src = &_sidata;
161 for (uint32_t *dst = &_sdata; (uintptr_t)dst < (uintptr_t)&_edata; ) {
162 *dst++ = *src++;
163 }
164
165 /* Zero .bss */
166 for (uint32_t *p = &_sbss; (uintptr_t)p < (uintptr_t)&_ebss; ) {
167 *p++ = 0;
168 }
169
170 main();
171 while (1) {}
172}
