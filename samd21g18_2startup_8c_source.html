1/*
2 * KTOS — ATSAMD21G18 (Arduino Zero) minimal bare-metal startup
3 *
4 * Provides:
5 * - Vector table (.vectors at 0x00002000, after 8 KB bootloader)
6 * - Reset_Handler: 48 MHz clock init, copy .data, zero .bss, call main()
7 * - Default_Handler: infinite loop for unhandled exceptions
8 * - TC3_Handler: declared weak — application main.c overrides it
9 *
10 * Clock setup: XOSC32K → GCLK1 → DFLL48M closed-loop → GCLK0 = 48 MHz.
11 * TC3 (IRQ 21) is the KTOS 1 ms tick source.
12 *
13 * Every main.c must define:
14 * void TC3_Handler(void) {
15 * *(volatile uint8_t *)0x42002C0EUL = 1; // clear TC3 INTFLAG.OVF
16 * ktos_timer_irq_handler();
17 * }
18 */

49static void clock_init_48MHz(void)
50{
51 /* 1 NVM wait state required above 24 MHz */
52 NVMCTRL_CTRLB = (NVMCTRL_CTRLB & ~(0xFu << 1)) | (1u << 1);
53
54 /* Enable XOSC32K crystal (32.768 kHz), 1 ms startup */
55 SYSCTRL_XOSC32K = (1u<<1)|(1u<<2)|(1u<<3)|(4u<<8); /* EN32K|XTALEN|STARTUP=4 */
56 SYSCTRL_XOSC32K |= (1u<<1); /* re-write with ENABLE bit set (bit 1 = ENABLE) */
57
58 /* Wait for XOSC32K ready */
59 while (!(SYSCTRL_PCLKSR & (1u << 0))) {}
60
61 /* GCLK1 = XOSC32K (32.768 kHz reference for DFLL) */
62 GCLK_GENDIV = (1u << 0); /* GCLK1 divisor = 1 */
63 while (GCLK_STATUS & (1u << 7)) {}
64 GCLK_GENCTRL = (1u << 0) | (5u << 8) | (1u << 16); /* ID=1|SRC=XOSC32K|GCLKEN */
65 while (GCLK_STATUS & (1u << 7)) {}
66
67 /* Connect GCLK1 to DFLL48M reference (peripheral clock ID 0) */
68 GCLK_CLKCTRL = (uint16_t)((1u << 8) | (1u << 14)); /* ID=0|GEN=1|CLKEN */
69 while (GCLK_STATUS & (1u << 7)) {}
70
71 /* Enable DFLL48M in open-loop first (required before switching to closed-loop) */
72 SYSCTRL_DFLLCTRL = 0;
73 while (!(SYSCTRL_PCLKSR & (1u << 4))) {} /* wait DFLLRDY */
74 SYSCTRL_DFLLCTRL = (1u << 1); /* ENABLE */
75 while (!(SYSCTRL_PCLKSR & (1u << 4))) {}
76
77 /* Configure closed-loop: MUL=1465 (48MHz/32768), FSTEP=10, CSTEP=1 */
78 SYSCTRL_DFLLMUL = (1465u) | (10u << 16) | (1u << 26);
79 while (SYSCTRL_DFLLSYNC & (1u << 7)) {}
80
81 /* Switch to closed-loop + WAITLOCK */
82 SYSCTRL_DFLLCTRL = (1u << 1) | (1u << 2) | (1u << 11); /* ENABLE|MODE|WAITLOCK */
83 while (!(SYSCTRL_PCLKSR & (1u << 4))) {} /* DFLLRDY */
84 while (!(SYSCTRL_PCLKSR & (1u << 6))) {} /* DFLLLCKF */
85
86 /* GCLK0 = DFLL48M (48 MHz system clock) */
87 GCLK_GENDIV = 0u; /* GCLK0 divisor = 1 */
88 while (GCLK_STATUS & (1u << 7)) {}
89 GCLK_GENCTRL = (7u << 8) | (1u << 16); /* ID=0|SRC=DFLL48M|GCLKEN */
90 while (GCLK_STATUS & (1u << 7)) {}
91}

180/* =========================================================================
181 * Reset handler
182 * ========================================================================= */
183void Reset_Handler(void)
184{
185 /* Relocate vector table to our application base */
186 SCB_VTOR = 0x00002000UL;
187
188 /* Bring system to 48 MHz */
189 clock_init_48MHz();
190
191 /* Copy .data from flash to RAM */
192 const uint32_t *src = &_sidata;
193 for (uint32_t *dst = &_sdata; (uintptr_t)dst < (uintptr_t)&_edata; ) { *dst++ = *src++; }
194
195 /* Zero .bss */
196 for (uint32_t *p = &_sbss; (uintptr_t)p < (uintptr_t)&_ebss; ) { *p++ = 0; }
197
198 main();
199 while (1) {}
200}
