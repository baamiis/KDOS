129void *ktos_hal_InitTaskStack(void *p_stack_base,
130 unsigned int stack_size_bytes,
131 WORD (*task_func_addr)(WORD, WORD, LONG),
132 void (*task_exit_handler_addr)(WORD),
133 WORD initial_msg_type,
134 WORD initial_sparam,
135 LONG initial_lparam)
136{
137 /* word-aligned */
138 uint16_t *sp = (uint16_t *)((uint8_t *)p_stack_base + stack_size_bytes);
139 sp = (uint16_t *)((uintptr_t)sp & ~(uintptr_t)1);
140
141 *--sp = (uint16_t)(uintptr_t)task_exit_handler_addr;
142 *--sp = (uint16_t)(uintptr_t)ktos_task_launch;
143 *--sp = 0; /* R11 */
144 *--sp = 0; /* R10 */
145 *--sp = 0; /* R9 */
146 *--sp = (uint16_t)(uintptr_t)task_func_addr; /* R8 */
147 *--sp = (uint16_t)((uint32_t)initial_lparam >> 16); /* R7 = lParam high */
148 *--sp = (uint16_t)((uint32_t)initial_lparam & 0xFFFF); /* R6 = lParam low */
149 *--sp = (uint16_t)initial_sparam; /* R5 */
150 *--sp = (uint16_t)initial_msg_type; /* R4 */
151 *--sp = 0x0008U; /* SR with GIE */
152
153 return (void *)sp;
154}

167__attribute__((naked)) void ktos_hal_ContextSwitch(
168 void **p_current_sp_storage __attribute__((unused)),
169 const void *next_sp __attribute__((unused)))
170{
171 __asm__ volatile (
172 "push r4 \n"
173 "push r5 \n"
174 "push r6 \n"
175 "push r7 \n"
176 "push r8 \n"
177 "push r9 \n"
178 "push r10 \n"
179 "push r11 \n"
180 "push r2 \n" /* SR */
181 "mov r1, 0(r12) \n" /* *p_current_sp_storage = SP */
182 "mov r13, r1 \n" /* SP = next_sp */
183 "pop r2 \n" /* restore SR */
184 "pop r11 \n"
185 "pop r10 \n"
186 "pop r9 \n"
187 "pop r8 \n"
188 "pop r7 \n"
189 "pop r6 \n"
190 "pop r5 \n"
191 "pop r4 \n"
192 "ret \n"
193 );
194}

202__attribute__((naked)) void ktos_hal_StartScheduler(
203 const void *first_task_sp __attribute__((unused)))
204{
205 __asm__ volatile (
206 "mov r12, r1 \n" /* SP = first_task_sp */
207 "pop r2 \n" /* restore SR */
208 "pop r11 \n"
209 "pop r10 \n"
210 "pop r9 \n"
211 "pop r8 \n"
212 "pop r7 \n"
213 "pop r6 \n"
214 "pop r5 \n"
215 "pop r4 \n"
216 "ret \n" /* jump to ktos_task_launch */
217 );
