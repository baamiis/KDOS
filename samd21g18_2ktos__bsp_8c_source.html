56/* =========================================================================
57 * System timer — TC3, 1 ms tick at 48 MHz
58 *
59 * Clock path: GCLK0 (48 MHz) → PM APBCMASK TC3 → TC3 MFRQ, DIV1.
60 * CC0 = 47999 → (47999+1) / 48000000 = 1 ms.
61 *
62 * TC3 NVIC IRQ = 18 (peripheral IRQ offset in SAMD21G18).
63 * GCLK peripheral ID for TC3 = 18 (shared with TCC2 → GCLK ID 18).
64 * ========================================================================= */
65#define PM_APBCMASK (*(volatile uint32_t *)0x40000420UL)
66#define GCLK_CLKCTRL (*(volatile uint16_t *)0x40000C02UL)
67#define GCLK_STATUS (*(volatile uint8_t *)0x40000C01UL)
68
69#define TC3_CTRLA (*(volatile uint16_t *)0x42002C00UL)
70#define TC3_CTRLBSET (*(volatile uint8_t *)0x42002C05UL)
71#define TC3_INTENCLR (*(volatile uint8_t *)0x42002C0CUL)
72#define TC3_INTENSET (*(volatile uint8_t *)0x42002C0DUL)
73#define TC3_INTFLAG (*(volatile uint8_t *)0x42002C0EUL)
74#define TC3_STATUS (*(volatile uint8_t *)0x42002C0FUL)
75#define TC3_CC0 (*(volatile uint16_t *)0x42002C18UL)
76
77#define NVIC_ISER0 (*(volatile uint32_t *)0xE000E100UL)
78
79void ktos_hal_InitSystemTimer(void (*timer_isr_addr)(void))
80{
81 /* Enable TC3 APB clock (APBCMASK bit 11) */
82 PM_APBCMASK |= (1u << 11);
83
84 /* Connect GCLK0 (48 MHz) to TC3 (GCLK peripheral ID 18 = TCC2/TC3) */
85 GCLK_CLKCTRL = (uint16_t)(18u | (1u << 14)); /* ID=18|GEN=0|CLKEN */
86 while (GCLK_STATUS & (1u << 7)) {}
87
88 /* Software reset TC3 */
89 TC3_CTRLA = (1u << 0); /* SWRST */
90 while (TC3_STATUS & (1u << 7)) {} /* wait SYNCBUSY */
91 while (TC3_CTRLA & (1u << 0)) {} /* wait SWRST self-clear */
92
93 /* 16-bit MFRQ mode, no prescaler (DIV1) */
94 TC3_CTRLA = (1u << 8); /* WAVEGEN=MFRQ; MODE=16-bit, PRESCALER=DIV1 are zero (default) */
95 while (TC3_STATUS & (1u << 7)) {}
96
97 /* CC0 = 47999 → 1 ms at 48 MHz */
98 TC3_CC0 = 47999u;
99 while (TC3_STATUS & (1u << 7)) {}
100
101 /* Enable OVF (overflow) interrupt */
102 TC3_INTENCLR = 0xFF;
103 TC3_INTENSET = (1u << 0); /* OVF */
104
105 /* Enable TC3 IRQ in NVIC (IRQ 18) */
106 NVIC_ISER0 = (1u << 18);
107
108 /* Enable TC3 */
109 TC3_CTRLA |= (1u << 1); /* ENABLE */
110 while (TC3_STATUS & (1u << 7)) {}
111
112 (void)timer_isr_addr;
113}

115/* =========================================================================
116 * Task stack initialisation — Cortex-M0+
117 *
118 * Stack layout (SP at lowest address after init):
119 * [R8][R9][R10][R11] ← software-saved, popped first into R4-R7 then moved
120 * [R4][R5][R6 ][R7 ] ← software-saved, popped directly
121 * [R0][R1][R2 ][R3 ] ← hardware exception frame (task args)
122 * [R12][LR][PC][xPSR] ← hardware exception frame
123 * ========================================================================= */
124void *ktos_hal_InitTaskStack(void *p_stack_base,
125 unsigned int stack_size_bytes,
126 WORD (*task_func_addr)(WORD, WORD, LONG),
127 void (*task_exit_handler_addr)(WORD),
128 WORD initial_msg_type,
129 WORD initial_sparam,
130 LONG initial_lparam)
131{
132 uint32_t *sp = (uint32_t *)((uint8_t *)p_stack_base + stack_size_bytes);
133 sp = (uint32_t *)((uint32_t)sp & ~0x7UL);
134
135 /* Hardware exception frame (high address → low) */
136 *--sp = 0x01000000UL; /* xPSR — Thumb bit */
137 *--sp = (uint32_t)task_func_addr; /* PC */
138 *--sp = (uint32_t)task_exit_handler_addr; /* LR */
139 *--sp = 0; /* R12 */
140 *--sp = 0; /* R3 */
141 *--sp = (uint32_t)((initial_lparam >> 16) & 0xFFFF);/* R2 */
142 *--sp = (uint32_t)(initial_sparam); /* R1 */
143 *--sp = (uint32_t)(initial_msg_type); /* R0 */
144
145 /* Software-saved registers (M0+ order: R4-R7 above R8-R11 on stack) */
146 *--sp = 0; /* R7 */
147 *--sp = 0; /* R6 */
148 *--sp = 0; /* R5 */
149 *--sp = 0; /* R4 */
150 *--sp = 0; /* R11 */
151 *--sp = 0; /* R10 */
152 *--sp = 0; /* R9 */
153 *--sp = 0; /* R8 */ /* ← SP */
154
155 return (void *)sp;
156}

158/* =========================================================================
159 * Context switch — Cortex-M0+ assembly
160 *
161 * M0+ cannot push/pop R8-R11 directly; move them through R4-R7.
162 * Push order: R4-R7 first (to higher address), then R8-R11 via R4-R7 (lower).
163 * Stack at *p_current_sp_storage: [R8][R9][R10][R11][R4][R5][R6][R7]
164 * ========================================================================= */
165__attribute__((naked)) void ktos_hal_ContextSwitch(
166 void **p_current_sp_storage __attribute__((unused)),
167 const void *next_sp __attribute__((unused)))
168{
169 __asm volatile (
170 /* Save R4-R7 (go to higher stack addresses) */
171 "push {r4, r5, r6, r7} \n"
172 /* Save R8-R11 via R4-R7 (go to lower stack addresses) */
173 "mov r4, r8 \n"
174 "mov r5, r9 \n"
175 "mov r6, r10 \n"
176 "mov r7, r11 \n"
177 "push {r4, r5, r6, r7} \n"
178 /* Store current SP */
179 "str sp, [r0] \n"
180 /* Load next SP */
181 "mov sp, r1 \n"
182 /* Restore R8-R11 of next context */
183 "pop {r4, r5, r6, r7} \n"
184 "mov r8, r4 \n"
185 "mov r9, r5 \n"
186 "mov r10, r6 \n"
187 "mov r11, r7 \n"
188 /* Restore R4-R7 of next context */
189 "pop {r4, r5, r6, r7} \n"
190 "bx lr \n"
191 );
192}

194/* =========================================================================
195 * Scheduler launch — Cortex-M0+
196 * ========================================================================= */
197__attribute__((naked)) void ktos_hal_StartScheduler(const void *first_task_sp __attribute__((unused)))
198{
199 __asm volatile (
200 "mov sp, r0 \n"
201 /* Restore R8-R11 */
202 "pop {r4, r5, r6, r7} \n"
203 "mov r8, r4 \n"
204 "mov r9, r5 \n"
205 "mov r10, r6 \n"
206 "mov r11, r7 \n"
207 /* Restore R4-R7 */
208 "pop {r4, r5, r6, r7} \n"
209 /* Pop hardware frame: R0-R3 */
210 "pop {r0, r1, r2, r3} \n"
211 /* Skip R12, LR; jump to PC */
212 "pop {r4} \n" /* discard R12 */
213 "pop {r4} \n" /* discard LR */
214 "pop {r4} \n" /* PC = task entry */
215 "bx r4 \n"
216 );
217}
