154void ktos_hal_InitSystemTimer(void (*timer_isr_addr)(void))
155{
156 /* Stop Timer0, configure 16-bit, prescaler 1:4, internal clock */
157 T0CON_sfr = 0x01U; /* T0CS=0 (Fosc/4), PSA=0 (prescaler on), T0PS=001 (1:4), TMR0ON=0, T08BIT=0 */
158
159 /* Pre-load for 1 ms period */
160 TMR0H_sfr = 0xF4U;
161 TMR0L_sfr = 0x48U;
162
163 /* Enable Timer0 overflow interrupt (bit5 of INTCON) */
164 INTCON_sfr |= (1 << 5); /* TMR0IE */
165
166 /* Start Timer0 */
167 T0CON_sfr |= (1 << 7); /* TMR0ON */
168
169 (void)timer_isr_addr;
170 /*
171 * ISR wiring — add to application:
172 * void interrupt high_priority isr(void) {
173 * if (INTCON_sfr & (1 << 2)) { // TMR0IF
174 * TMR0H_sfr = 0xF4; TMR0L_sfr = 0x48; // reload
175 * INTCON_sfr &= ~(1 << 2); // clear TMR0IF
176 * ktos_timer_irq_handler();
177 * }
178 * }
179 */
180}

199void *ktos_hal_InitTaskStack(void *p_stack_base,
200 unsigned int stack_size_bytes,
201 WORD (*task_func_addr)(WORD, WORD, LONG),
202 void (*task_exit_handler_addr)(WORD),
203 WORD initial_msg_type,
204 WORD initial_sparam,
205 LONG initial_lparam)
206{
207 /* Place context block at the base of the provided buffer */
208 ktos_pic18_ctx_t *ctx = (ktos_pic18_ctx_t *)p_stack_base;
209
210 if (stack_size_bytes < sizeof(ktos_pic18_ctx_t)) {
211 return (void *)0;
212 }
213
214 ctx->wreg = 0;
215 ctx->status = 0;
216 ctx->bsr = 0;
217 ctx->fsr0 = 0;
218 /* FSR1 points to just above the context block — SDCC software stack base */
219 ctx->fsr1 = (uint16_t)((uintptr_t)p_stack_base + sizeof(ktos_pic18_ctx_t));
220 ctx->fsr2 = ctx->fsr1;
221 ctx->prodl = (uint8_t)(initial_msg_type & 0xFF);
222 ctx->prodh = (uint8_t)((initial_msg_type >> 8) & 0xFF);
223 ctx->pc = (uint16_t)(uintptr_t)task_func_addr;
224
225 (void)task_exit_handler_addr;
226 (void)initial_sparam;
227 (void)initial_lparam;
228
229 return (void *)ctx;
230}

246void ktos_hal_ContextSwitch(void **p_current_sp_storage, const void *next_sp)
247{
248 ktos_pic18_ctx_t *cur = (ktos_pic18_ctx_t *)*p_current_sp_storage;
249 ktos_pic18_ctx_t *next = (ktos_pic18_ctx_t *)next_sp;
250
251 /* Save current context */
252 cur->fsr1 = (uint16_t)((uint8_t)FSR1L_sfr | ((uint8_t)FSR1H_sfr << 8));
253 cur->fsr0 = (uint16_t)((uint8_t)FSR0L_sfr | ((uint8_t)FSR0H_sfr << 8));
254 cur->fsr2 = 0; /* FSR2 not used in this minimal save */
255 cur->prodl = PRODL_sfr;
256 cur->prodh = PRODH_sfr;
257 cur->bsr = BSR_sfr;
258 /* WREG and STATUS are restored by the scheduler on next dispatch */
259
260 /* Restore next context */
261 FSR1L_sfr = (uint8_t)(next->fsr1 & 0xFF);
262 FSR1H_sfr = (uint8_t)(next->fsr1 >> 8);
263 FSR0L_sfr = (uint8_t)(next->fsr0 & 0xFF);
264 FSR0H_sfr = (uint8_t)(next->fsr0 >> 8);
265 PRODL_sfr = next->prodl;
266 PRODH_sfr = next->prodh;
267 BSR_sfr = next->bsr;
268
269 /* Update p_current_sp_storage to point to new context for return */
270 *p_current_sp_storage = (void *)next;
271}

284void ktos_hal_StartScheduler(const void *first_task_sp)
285{
286 const ktos_pic18_ctx_t *ctx = (const ktos_pic18_ctx_t *)first_task_sp;
287
288 FSR1L_sfr = (uint8_t)(ctx->fsr1 & 0xFF);
289 FSR1H_sfr = (uint8_t)(ctx->fsr1 >> 8);
290 FSR0L_sfr = (uint8_t)(ctx->fsr0 & 0xFF);
291 FSR0H_sfr = (uint8_t)(ctx->fsr0 >> 8);
292 BSR_sfr = ctx->bsr;
293
294 /* Interrupts enabled before first task runs */
295 ktos_hal_EnableInterrupts();
296
297 /*
298 * The scheduler (ktos.c) must invoke the first task function directly
299 * after this returns, using the function pointer stored during InitTask.
300 * The PC stored in ctx->pc is informational only — PIC18 cannot set PC
301 * directly from C.
302 */
303}
