29void *ktos_hal_InitTaskStack(void *p_stack_base,
30 unsigned int stack_size_bytes,
31 WORD (*task_func_addr)(WORD, WORD, LONG),
32 void (*task_exit_handler_addr)(WORD),
33 WORD initial_msg_type,
34 WORD initial_sparam,
35 LONG initial_lparam)
36{
37 if (stack_size_bytes < 72u) return (void *)0;
38
39 uint32_t *sp = (uint32_t *)((uint8_t *)p_stack_base + stack_size_bytes);
40 /* Align to 8 bytes */
41 sp = (uint32_t *)((uint32_t)sp & ~7u);
42
43 /* --- Hardware-saved frame (auto pushed on exception entry) --- */
44 /* xPSR */
45 sp--; *sp = 0x01000000u;
46 /* PC = task entry */
47 sp--; *sp = ((uint32_t)task_func_addr) | 1u;
48 /* LR = exit handler */
49 sp--; *sp = ((uint32_t)task_exit_handler_addr) | 1u;
50 /* R12 */
51 sp--; *sp = 0x00000000u;
52 /* R3 = initial_lparam (upper half unused on M4, pass as R3:R2 pair) */
53 sp--; *sp = (uint32_t)(initial_lparam >> 16);
54 /* R2 = initial_lparam low */
55 sp--; *sp = (uint32_t)(initial_lparam & 0xFFFFu);
56 /* R1 = initial_sparam */
57 sp--; *sp = (uint32_t)initial_sparam;
58 /* R0 = initial_msg_type */
59 sp--; *sp = (uint32_t)initial_msg_type;
60
61 /* --- Software-saved frame (r4-r11) --- */
62 sp--; *sp = 0x11111111u; /* R11 */
63 sp--; *sp = 0x10101010u; /* R10 */
64 sp--; *sp = 0x09090909u; /* R9 */
65 sp--; *sp = 0x08080808u; /* R8 */
66 sp--; *sp = 0x07070707u; /* R7 */
67 sp--; *sp = 0x06060606u; /* R6 */
68 sp--; *sp = 0x05050505u; /* R5 */
69 sp--; *sp = 0x04040404u; /* R4 */
70
71 return (void *)sp;
72}
