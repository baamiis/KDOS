105 void *ktos_hal_InitTaskStack(void *p_stack_base,
106  unsigned int stack_size_bytes,
107  WORD (*task_func_addr)(WORD, WORD, LONG),
108  void (*task_exit_handler_addr)(WORD),
109  WORD initial_msg_type,
110  WORD initial_sparam,
111  LONG initial_lparam)
112 {
113  uint32_t *sp = (uint32_t *)((uint8_t *)p_stack_base + stack_size_bytes);
114  sp = (uint32_t *)((uint32_t)sp & ~0x7UL);
115 
116  *--sp = 0x01000000UL;
117  *--sp = (uint32_t)task_func_addr;
118  *--sp = (uint32_t)task_exit_handler_addr;
119  *--sp = 0;
120  *--sp = 0;
121  *--sp = (uint32_t)((initial_lparam >> 16) & 0xFFFF);
122  *--sp = (uint32_t)(initial_sparam);
123  *--sp = (uint32_t)(initial_msg_type);
124 
125  /* R4-R11 */
126  for (int i = 0; i < 8; i++) *--sp = 0;
127 
128  return (void *)sp;
129 }

140 __attribute__((naked)) void ktos_hal_ContextSwitch(
141  void **p_current_sp_storage __attribute__((unused)),
142  const void *next_sp __attribute__((unused)))
143 {
144  __asm volatile (
145  /* Cortex-M0: save R4-R7 then R8-R11 via low registers */
146  "push {r4-r7} \n"
147  "mov r4, r8 \n"
148  "mov r5, r9 \n"
149  "mov r6, r10 \n"
150  "mov r7, r11 \n"
151  "push {r4-r7} \n"
152  /* Cortex-M0 cannot STR SP directly — move SP to low reg first */
153  "mov r2, sp \n"
154  "str r2, [r0] \n" /* *p_current_sp_storage = SP */
155  "mov sp, r1 \n" /* SP = next_sp */
156  "pop {r4-r7} \n"
157  "mov r8, r4 \n"
158  "mov r9, r5 \n"
159  "mov r10, r6 \n"
160  "mov r11, r7 \n"
161  "pop {r4-r7} \n"
162  "bx lr \n"
163  );
164 }

174 __attribute__((naked)) void ktos_hal_StartScheduler(const void *first_task_sp __attribute__((unused)))
175 {
176  __asm volatile (

178  "pop {r4-r7} \n"
179  "mov r8, r4 \n"
180  "mov r9, r5 \n"
181  "mov r10, r6 \n"
182  "mov r11, r7 \n"
183  "pop {r4-r7} \n"
184  "pop {r0-r3} \n"
185  "pop {r4} \n" /* R12 */
186  "pop {r5} \n" /* LR */
187  "pop {r6} \n" /* PC */
188  "pop {r7} \n" /* xPSR - discard */
189  "bx r6 \n"
190  );
191 }
