136 void *ktos_hal_InitTaskStack(void *p_stack_base,
137  unsigned int stack_size_bytes,
138  WORD (*task_func_addr)(WORD, WORD, LONG),
139  void (*task_exit_handler_addr)(WORD),
140  WORD initial_msg_type,
141  WORD initial_sparam,
142  LONG initial_lparam)
143 {
144  uint32_t *sp = (uint32_t *)((uint8_t *)p_stack_base + stack_size_bytes);
145  sp = (uint32_t *)((uint32_t)sp & ~0x7UL); /* 8-byte alignment */
146 
147  /* Hardware exception frame */
148  *--sp = 0x01000000UL; /* xPSR — Thumb bit set */
149  *--sp = (uint32_t)task_func_addr; /* PC */
150  *--sp = (uint32_t)task_exit_handler_addr; /* LR */
151  *--sp = 0; /* R12 */
152  *--sp = 0; /* R3 */
153  *--sp = (uint32_t)((initial_lparam >> 16) & 0xFFFF); /* R2 */
154  *--sp = (uint32_t)(initial_sparam); /* R1 */
155  *--sp = (uint32_t)(initial_msg_type); /* R0 */
156 
157  /* Software-saved registers R4-R11 */
158  *--sp = 0; /* R11 */
159  *--sp = 0; /* R10 */
160  *--sp = 0; /* R9 */
161  *--sp = 0; /* R8 */
162  *--sp = 0; /* R7 */
163  *--sp = 0; /* R6 */
164  *--sp = 0; /* R5 */
165  *--sp = 0; /* R4 */
166 
167  return (void *)sp;
168 }

184 __attribute__((naked)) void ktos_hal_ContextSwitch(
185  void **p_current_sp_storage __attribute__((unused)),
186  const void *next_sp __attribute__((unused)))
187 {
188  __asm volatile (
189  "push {r4-r11} \n" /* Save callee-saved registers */
190  "str sp, [r0] \n" /* *p_current_sp_storage = SP */
191  "mov sp, r1 \n" /* SP = next_sp */
192  "pop {r4-r11} \n" /* Restore new context */
193  "bx lr \n" /* Return into new task */
194  );
195 }

211 {
