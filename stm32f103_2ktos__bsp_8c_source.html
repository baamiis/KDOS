84/* SysTick registers */
85#define SYSTICK_BASE 0xE000E010UL
86#define SYSTICK_CTRL (*(volatile uint32_t *)(SYSTICK_BASE + 0x00))
87#define SYSTICK_LOAD (*(volatile uint32_t *)(SYSTICK_BASE + 0x04))
88#define SYSTICK_VAL (*(volatile uint32_t *)(SYSTICK_BASE + 0x08))
89
90#define SYSTICK_CTRL_ENABLE (1 << 0)
91#define SYSTICK_CTRL_TICKINT (1 << 1)
92#define SYSTICK_CTRL_CLKSOURCE (1 << 2)
93
98void ktos_hal_InitSystemTimer(void (*timer_isr_addr)(void))
99{
100 /* 72MHz / 1000 = 72000 ticks per 1ms */
101 SYSTICK_LOAD = 72000 - 1;
102 SYSTICK_VAL = 0;
103 SYSTICK_CTRL = SYSTICK_CTRL_CLKSOURCE |
104 SYSTICK_CTRL_TICKINT |
105 SYSTICK_CTRL_ENABLE;
106 (void)timer_isr_addr;
107 /* SysTick_Handler in the vector table must call ktos_timer_irq_handler():
108 *
109 * void SysTick_Handler(void) { ktos_timer_irq_handler(); }
110 */
111}

136void *ktos_hal_InitTaskStack(void *p_stack_base,
137 unsigned int stack_size_bytes,
138 WORD (*task_func_addr)(WORD, WORD, LONG),
139 void (*task_exit_handler_addr)(WORD),
140 WORD initial_msg_type,
141 WORD initial_sparam,
142 LONG initial_lparam)
143{
144 uint32_t *sp = (uint32_t *)((uint8_t *)p_stack_base + stack_size_bytes);
145 sp = (uint32_t *)((uint32_t)sp & ~0x7UL); /* 8-byte alignment */
146
147 /* Hardware exception frame */
148 *--sp = 0x01000000UL; /* xPSR — Thumb bit set */
149 *--sp = (uint32_t)task_func_addr; /* PC */
150 *--sp = (uint32_t)task_exit_handler_addr; /* LR */
151 *--sp = 0; /* R12 */
152 *--sp = 0; /* R3 */
153 *--sp = (uint32_t)((initial_lparam >> 16) & 0xFFFF); /* R2 */
154 *--sp = (uint32_t)(initial_sparam); /* R1 */
155 *--sp = (uint32_t)(initial_msg_type); /* R0 */
156
157 /* Software-saved registers R4-R11 */
158 *--sp = 0; /* R11 */
159 *--sp = 0; /* R10 */
160 *--sp = 0; /* R9 */
161 *--sp = 0; /* R8 */
162 *--sp = 0; /* R7 */
163 *--sp = 0; /* R6 */
164 *--sp = 0; /* R5 */
165 *--sp = 0; /* R4 */
166
167 return (void *)sp;
168}

184__attribute__((naked)) void ktos_hal_ContextSwitch(
185 void **p_current_sp_storage __attribute__((unused)),
186 const void *next_sp __attribute__((unused)))
187{
188 __asm volatile (
189 "push {r4-r11} \n" /* Save callee-saved registers */
190 "str sp, [r0] \n" /* *p_current_sp_storage = SP */
191 "mov sp, r1 \n" /* SP = next_sp */
192 "pop {r4-r11} \n" /* Restore new context */
193 "bx lr \n" /* Return into new task */
194 );
195}

197/* =========================================================================
198 * Scheduler launch
199 * ========================================================================= */
200
210__attribute__((naked)) void ktos_hal_StartScheduler(const void *first_task_sp __attribute__((unused)))
211{
212 __asm volatile (
213 "mov sp, r0 \n" /* Set SP to first task stack */
214 "pop {r4-r11} \n" /* Restore R4-R11 */
215 "pop {r0-r3, r12, lr} \n" /* Restore R0-R3, R12, LR */
216 "pop {pc} \n" /* Jump to task (pop PC + xPSR) */
217 );
218}
