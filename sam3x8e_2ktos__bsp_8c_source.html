90/* =========================================================================
91 * System timer — TC0 channel 0, 1 ms tick at 84 MHz
92 *
93 * Why TC0 and not SysTick? On the SAM3X8E the Cortex-M3 SysTick is
94 * frequently claimed by upper-layer code (e.g. the Arduino-SAM core's
95 * `millis()`). Using a dedicated Timer/Counter peripheral lets KTOS
96 * coexist with any startup environment without fighting over a shared
97 * interrupt vector.
98 *
99 * Clock chain: MCK = 84 MHz → TC TIMER_CLOCK2 (= MCK/8) = 10.5 MHz.
100 * RC = 10 500 → compare match every 1.0 ms exactly.
101 *
102 * The application MUST define TC0_Handler() to call
103 * @c ktos_timer_irq_handler() and MUST read TC0->TC_CHANNEL[0].TC_SR
104 * to clear the compare flag (otherwise the interrupt re-fires
105 * immediately). See examples for the canonical wiring.
106 * ========================================================================= */
107
108/* Peripheral Management Controller (PMC) — for enabling TC0 clock. */
109#define SAM_PMC_PCER0 (*(volatile uint32_t *)0x400E0610UL)
110
111/* TC0 channel 0 register block (datasheet §36). */
112#define TC0_CH0_BASE 0x40080000UL
113#define TC0_CH0_CCR (*(volatile uint32_t *)(TC0_CH0_BASE + 0x00))
114#define TC0_CH0_CMR (*(volatile uint32_t *)(TC0_CH0_BASE + 0x04))
115#define TC0_CH0_RC (*(volatile uint32_t *)(TC0_CH0_BASE + 0x1C))
116#define TC0_CH0_SR (*(volatile uint32_t *)(TC0_CH0_BASE + 0x20))
117#define TC0_CH0_IER (*(volatile uint32_t *)(TC0_CH0_BASE + 0x24))
118#define TC0_CH0_IDR (*(volatile uint32_t *)(TC0_CH0_BASE + 0x28))
119
120/* TC_CCR (Channel Control). */
121#define TC_CCR_CLKEN (1u << 0)
122#define TC_CCR_CLKDIS (1u << 1)
123#define TC_CCR_SWTRG (1u << 2)
124
125/* TC_CMR (Channel Mode) — waveform mode, MCK/8, reset on RC. */
126#define TC_CMR_TCCLKS_TIMER_CLOCK2 (1u << 0) /* MCK/8 */
127#define TC_CMR_WAVSEL_UP_RC (2u << 13)
128#define TC_CMR_WAVE (1u << 15)
129
130/* TC interrupt: RC compare. */
131#define TC_IER_CPCS (1u << 4)
132
133/* Cortex-M3 NVIC — enable IRQ 27 (TC0 channel 0). */
134#define NVIC_ISER0 (*(volatile uint32_t *)0xE000E100UL)
135#define TC0_IRQ_NUMBER 27u
136
147void ktos_hal_InitSystemTimer(void (*timer_isr_addr)(void))
148{
149 /* Enable PMC clock to TC0 channel 0 (peripheral ID 27). */
150 SAM_PMC_PCER0 = (1u << TC0_IRQ_NUMBER);
151
152 /* Disable channel before reconfiguring. */
153 TC0_CH0_CCR = TC_CCR_CLKDIS;
154
155 /* Waveform mode, MCK/8, reset on RC match. */
156 TC0_CH0_CMR = TC_CMR_TCCLKS_TIMER_CLOCK2 |
157 TC_CMR_WAVE |
158 TC_CMR_WAVSEL_UP_RC;
159
160 /* 84 MHz / 8 = 10.5 MHz → 10 500 ticks per 1 ms. */
161 TC0_CH0_RC = 10500u;
162
163 /* Enable RC compare interrupt; mask everything else. */
164 TC0_CH0_IDR = 0xFFFFFFFFu;
165 TC0_CH0_IER = TC_IER_CPCS;
166
167 /* Enable TC0 channel 0 IRQ in the NVIC. */
168 NVIC_ISER0 = (1u << TC0_IRQ_NUMBER);
169
170 /* Enable the clock and software-trigger the counter. */
171 TC0_CH0_CCR = TC_CCR_CLKEN | TC_CCR_SWTRG;
172
173 (void)timer_isr_addr;
174}

199void *ktos_hal_InitTaskStack(void *p_stack_base,
200 unsigned int stack_size_bytes,
201 WORD (*task_func_addr)(WORD, WORD, LONG),
202 void (*task_exit_handler_addr)(WORD),
203 WORD initial_msg_type,
204 WORD initial_sparam,
205 LONG initial_lparam)
206{
207 uint32_t *sp = (uint32_t *)((uint8_t *)p_stack_base + stack_size_bytes);
208 sp = (uint32_t *)((uint32_t)sp & ~0x7UL); /* 8-byte alignment */
209
210 /* Hardware exception frame */
211 *--sp = 0x01000000UL; /* xPSR — Thumb bit set */
212 *--sp = (uint32_t)task_func_addr; /* PC */
213 *--sp = (uint32_t)task_exit_handler_addr; /* LR */
214 *--sp = 0; /* R12 */
215 *--sp = 0; /* R3 */
216 *--sp = (uint32_t)((initial_lparam >> 16) & 0xFFFF); /* R2 */
217 *--sp = (uint32_t)(initial_sparam); /* R1 */
218 *--sp = (uint32_t)(initial_msg_type); /* R0 */
219
220 /* Software-saved registers R4-R11 */
221 *--sp = 0; /* R11 */
222 *--sp = 0; /* R10 */
223 *--sp = 0; /* R9 */
224 *--sp = 0; /* R8 */
225 *--sp = 0; /* R7 */
226 *--sp = 0; /* R6 */
227 *--sp = 0; /* R5 */
228 *--sp = 0; /* R4 */
229
230 return (void *)sp;
231}

246__attribute__((naked)) void ktos_hal_ContextSwitch(
247 void **p_current_sp_storage __attribute__((unused)),
248 const void *next_sp __attribute__((unused)))
249{
250 __asm volatile (
251 "push {r4-r11} \n" /* Save callee-saved registers */
252 "str sp, [r0] \n" /* *p_current_sp_storage = SP */
253 "mov sp, r1 \n" /* SP = next_sp */
254 "pop {r4-r11} \n" /* Restore new context */
255 "bx lr \n" /* Return into new task */
256 );
257}

271__attribute__((naked)) void ktos_hal_StartScheduler(const void *first_task_sp __attribute__((unused)))
272{
273 __asm volatile (
274 "mov sp, r0 \n" /* Set SP to first task stack */
275 "pop {r4-r11} \n" /* Restore R4-R11 */
276 "pop {r0-r3, r12, lr} \n" /* Restore R0-R3, R12, LR */
277 "pop {pc} \n" /* Jump to task (pop PC + xPSR) */
278 );
279}
