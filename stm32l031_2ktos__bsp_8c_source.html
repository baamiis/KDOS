107void *ktos_hal_InitTaskStack(void *p_stack_base,
108 unsigned int stack_size_bytes,
109 WORD (*task_func_addr)(WORD, WORD, LONG),
110 void (*task_exit_handler_addr)(WORD),
111 WORD initial_msg_type,
112 WORD initial_sparam,
113 LONG initial_lparam)
114{
115 uint32_t *sp = (uint32_t *)((uint8_t *)p_stack_base + stack_size_bytes);
116 sp = (uint32_t *)((uint32_t)sp & ~0x7UL);
117
118 *--sp = 0x01000000UL; /* xPSR: Thumb bit */
119 *--sp = (uint32_t)task_func_addr; /* PC */
120 *--sp = (uint32_t)task_exit_handler_addr; /* LR */
121 *--sp = 0; /* R12 */
122 *--sp = 0; /* R3 */
123 *--sp = (uint32_t)((initial_lparam >> 16) & 0xFFFF); /* R2 = lParam high */
124 *--sp = (uint32_t)(initial_sparam); /* R1 */
125 *--sp = (uint32_t)(initial_msg_type); /* R0 */
126
127 /* R4-R11 */
128 for (int i = 0; i < 8; i++) *--sp = 0;
129
130 return (void *)sp;
131}

139__attribute__((naked)) void ktos_hal_ContextSwitch(
140 void **p_current_sp_storage __attribute__((unused)),
141 const void *next_sp __attribute__((unused)))
142{
143 __asm volatile (
144 "push {r4-r7} \n"
145 "mov r4, r8 \n"
146 "mov r5, r9 \n"
147 "mov r6, r10 \n"
148 "mov r7, r11 \n"
149 "push {r4-r7} \n"
150 "mov r2, sp \n"
151 "str r2, [r0] \n" /* *p_current_sp_storage = SP */
152 "mov sp, r1 \n" /* SP = next_sp */
153 "pop {r4-r7} \n"
154 "mov r8, r4 \n"
155 "mov r9, r5 \n"
156 "mov r10, r6 \n"
157 "mov r11, r7 \n"
158 "pop {r4-r7} \n"
159 "bx lr \n"
160 );
161}

168__attribute__((naked)) void ktos_hal_StartScheduler(const void *first_task_sp __attribute__((unused)))
169{
170 __asm volatile (
171 "mov sp, r0 \n"
172 "pop {r4-r7} \n"
173 "mov r8, r4 \n"
174 "mov r9, r5 \n"
175 "mov r10, r6 \n"
176 "mov r11, r7 \n"
177 "pop {r4-r7} \n"
178 "pop {r0-r3} \n"
179 "pop {r4} \n" /* R12 */
180 "pop {r5} \n" /* LR */
181 "pop {r6} \n" /* PC */
182 "pop {r7} \n" /* xPSR - discard */
183 "bx r6 \n"
184 );
185}
