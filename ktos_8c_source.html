118static void ktos_DefaultTaskExitHandler(WORD task_return_value)
119{
120#if DEBUG
121 if (TaskCurrent) {
122 ktos_DebugPrintf("Task '%c' exited with value %u.\n",
123 TaskCurrent->TaskID, task_return_value);
124 } else {
125 ktos_DebugPrintf("Unknown task exited.\n");
126 }
127#endif
128
129 g_LastTaskReturnValue = task_return_value;
130 TaskCurrent->NeedsReinit = TRUE;
131
132 /* Return to the OS scheduler context. */
133 ktos_hal_ContextSwitch((void **)&(TaskCurrent->StackPtr), OS_SP);
134
135 ktos_Emergency("ExitHandler_CtxSwitch_Failed");
136}

142struct ktos_TASK *ktos_InitTask(
143 WORD (*Func)(WORD MsgType, WORD sParam, LONG lParam),
144 INT StackSize,
145 INT QueueSize,
146 BYTE TaskIDVal)
147{
148 struct ktos_TASK *Task;
149 int32_t *Stack;
150
151 Task = (struct ktos_TASK *)malloc(sizeof(struct ktos_TASK));
152 if (Task == NULL) { ktos_Emergency("T Failed"); }
153
154 Stack = (int32_t *)calloc(StackSize, sizeof(int32_t));
155 if (Stack == NULL) { ktos_Emergency("S Failed"); }
156
157 Task->MsgQueue = (struct ktos_MSG *)calloc(QueueSize, sizeof(struct ktos_MSG));
158 if (Task->MsgQueue == NULL) { ktos_Emergency("Q Failed"); }
159
160 Task->Func = Func;
161 Task->TaskID = TaskIDVal;
162 Task->QueueCapacity = QueueSize;
163 Task->StackBase = Stack;
164 Task->StackSizeBytes = (unsigned int)((unsigned int)StackSize * sizeof(int32_t));
165 Task->NeedsReinit = FALSE;
166
167 Task->StackPtr = ktos_hal_InitTaskStack(
168 Stack,
169 StackSize * sizeof(int32_t),
170 Func,
171 ktos_DefaultTaskExitHandler,
172 KTOS_MSG_TYPE_INIT,
173 (WORD)0,
174 (LONG)0L);
175
176 if (Task->StackPtr == NULL) { ktos_Emergency("StackInit Failed"); }
177
178 Task->MsgQueueIn = Task->MsgQueue;
179 Task->MsgQueueOut = Task->MsgQueue;
180 Task->MsgQueueEnd = Task->MsgQueue + QueueSize;
181 Task->Timer = 0;
182 Task->TimerFlag = TRUE; /* ready for initial INIT dispatch */
183 Task->Sleeping = FALSE;
184 Task->MsgCount = 0;
185
186 /* Insert into the circular task ring. */
187 if (TaskCurrent == NULL) {
188 Task->TaskNext = Task;
189 } else {
190 Task->TaskNext = TaskCurrent->TaskNext;
191 TaskCurrent->TaskNext = Task;
192 }
193 TaskCurrent = Task;
194
195 return Task;
196}

213bool ktos_SendMsg(struct ktos_TASK *Task,
214 WORD MsgType,
215 WORD sParam,
216 LONG lParam)
217{
218 if (Task) {
219 if (Task->MsgCount >= Task->QueueCapacity) { return false; }
220 ktos_hal_DisableInterrupts();
221 struct ktos_MSG *Msg = Task->MsgQueueIn;
222 Msg->MsgType = MsgType;
223 Msg->sParam = sParam;
224 Msg->lParam = lParam;
225 if (++Task->MsgQueueIn >= Task->MsgQueueEnd) {
226 Task->MsgQueueIn = Task->MsgQueue;
227 }
228 ++Task->MsgCount;
229 ktos_hal_EnableInterrupts();
230 return true;
231 }
232 return false;
233}

269static void ktos_SwitchTask(void)
270{
271 static WORD Delay;
272
273 while (TRUE)
274 {
275 ktos_hal_DisableInterrupts();
276
277 if (ktos_exit_requested) {
278 ktos_hal_EnableInterrupts();
279 return;
280 }
281
282 if (MultiTask) {
283 TaskCurrent = TaskCurrent->TaskNext;
284 }
285
286 if (TaskCurrent->Sleeping) {
287 if (TaskCurrent->TimerFlag) {
288 /* Sleep timer expired — mark the task ready. */
289 TaskCurrent->Timer = 0;
290 TaskCurrent->TimerFlag = FALSE;
291 TaskCurrent->Sleeping = FALSE;
292 }
293 }
294
295 if (!TaskCurrent->Sleeping &&
296 (TaskCurrent->MsgCount != 0 || TaskCurrent->TimerFlag))
297 {
298 if (TaskCurrent->NeedsReinit) {
299 /*
300 * Handler model: task previously returned a value.
301 * Re-initialise the stack with the next pending message so
302 * the task is called fresh as task_func(MsgType, sParam, lParam).
303 */
304 WORD msg_type = KTOS_MSG_TYPE_TIMER;
305 WORD s_param = 0;
306 LONG l_param = 0;
307
308 if (TaskCurrent->MsgCount != 0) {
309 /* Read message content before consuming. */
310 msg_type = TaskCurrent->MsgQueueOut->MsgType;
311 s_param = TaskCurrent->MsgQueueOut->sParam;
312 l_param = TaskCurrent->MsgQueueOut->lParam;
313 if (++TaskCurrent->MsgQueueOut >= TaskCurrent->MsgQueueEnd) {
314 TaskCurrent->MsgQueueOut = TaskCurrent->MsgQueue;
315 }
316 --TaskCurrent->MsgCount;
317 }
318
319 TaskCurrent->NeedsReinit = FALSE;
320 TaskCurrent->StackPtr = (int32_t *)ktos_hal_InitTaskStack(
321 TaskCurrent->StackBase,
322 TaskCurrent->StackSizeBytes,
323 TaskCurrent->Func,
324 ktos_DefaultTaskExitHandler,
325 msg_type, s_param, l_param);
326 }
327 /* else: coroutine model (ktos_Sleep) or fresh first-run — restore
328 * existing StackPtr as-is; do not consume from the message queue. */
329
330 /* Switch into the task; returns when the task yields back. */
331 ktos_hal_ContextSwitch((void **)&OS_SP, TaskCurrent->StackPtr);
332 /* Interrupts assumed disabled on return. */
333
334 Delay = g_LastTaskReturnValue;
335
336 if (Delay == 0) {
337 /* Yield — re-schedule immediately. */
338 TaskCurrent->TimerFlag = TRUE;
339 TaskCurrent->Timer = 0;
340 } else if (Delay == MSG_WAIT) {
341 /* Sleep indefinitely until a message arrives. */
342 TaskCurrent->Timer = 0;
343 TaskCurrent->TimerFlag = FALSE;
344 } else {
345 /* Sleep for Delay milliseconds. */
346 TaskCurrent->Timer = Delay;
347 TaskCurrent->TimerFlag = FALSE;
348 }
349 }
350
351 ktos_hal_EnableInterrupts();
352 }
353}

355INT ktos_Sleep(WORD Delay, bool TaskSwitchPermit)
356{
357 ktos_hal_DisableInterrupts();
358
359 TaskCurrent->Sleeping = TRUE;
360 TaskCurrent->WakeUpType = 0;
361
362 if (Delay == 0) {
363 TaskCurrent->TimerFlag = TRUE;
364 TaskCurrent->Timer = 0;
365 } else if (Delay == MSG_WAIT) {
366 TaskCurrent->Timer = 0;
367 } else {
368 TaskCurrent->Timer = Delay;
369 }
370
371 MultiTask = TaskSwitchPermit;
372
373 ktos_hal_ContextSwitch((void **)&(TaskCurrent->StackPtr), OS_SP);
374
375 MultiTask = TRUE;
376 ktos_hal_EnableInterrupts();
377 return TaskCurrent->WakeUpType;
378}

395void ktos_timer_irq_handler(void)
396{
397 struct ktos_TASK *Task = TaskCurrent;
398 if (!Task) return;
399 do {
400 if (Task->Timer) {
401 if (--Task->Timer == 0) {
402 Task->TimerFlag = TRUE;
403 }
404 }
405 Task = Task->TaskNext;
406 } while (Task != TaskCurrent);
407}
