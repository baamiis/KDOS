75void *ktos_hal_InitTaskStack(void *p_stack_base,
76 unsigned int stack_size_bytes,
77 void (*task_func_addr)(WORD, WORD, LONG),
78 void (*task_exit_handler_addr)(WORD),
79 WORD initial_msg_type,
80 WORD initial_sparam,
81 LONG initial_lparam)
82{
83 // TODO: Implement this function to set up the initial stack frame for a new task.
84 // This is highly architecture-specific and usually involves some assembly.
85 //
86 // What this function needs to do:
87 // 1. Calculate the actual top-of-stack (TOS) pointer. Stacks usually grow downwards.
88 // So, TOS would be something like `(char*)p_stack_base + stack_size_bytes`.
89 // Ensure the stack pointer is correctly aligned for the CPU architecture.
90 //
91 // 2. "Push" initial CPU register values onto this stack in the order the CPU expects
92 // them for a context restore. This includes:
93 // - A Program Status Register (PSR) with appropriate mode/interrupt settings.
94 // - The Program Counter (PC) set to `task_func_addr`.
95 // - The Link Register (LR) set to `task_exit_handler_addr`.
96 // - Initial values for general-purpose registers (often 0 or a debug pattern).
97 // - Registers for passing parameters: `initial_msg_type`, `initial_sparam`, `initial_lparam`
98 // must be placed where the `task_func_addr` will find them according to the
99 // architecture's calling convention (e.g., R0, R1, R2 on ARM).
100 //
101 // 3. The `task_exit_handler_addr` must be set up so that if `task_func_addr` returns,
102 // the `task_exit_handler_addr` is called, and the `WORD` return value from
103 // `task_func_addr` is passed as the first argument to `task_exit_handler_addr`.
104 // This might involve setting LR to a small assembly trampoline that moves the
105 // return value from its register (e.g., R0) into the first argument register for
106 // `task_exit_handler_addr` and then branches to `task_exit_handler_addr`.
107 //
108 // 4. Return the final value of the stack pointer (after all initial context is pushed).
109 // This will be the value stored in `Task->StackPointer`.
110 //
111 // Example (conceptual for ARM Cortex-M - actual implementation is more complex):
112 // unsigned long *stack_ptr = (unsigned long*)(((unsigned char*)p_stack_base + stack_size_bytes) & ~0x7UL); // Ensure 8-byte alignment
113 // *(--stack_ptr) = 0x01000000L; // xPSR (Thumb state)
114 // *(--stack_ptr) = (unsigned long)task_func_addr; // PC
115 // *(--stack_ptr) = (unsigned long)task_exit_handler_addr; // LR (or trampoline that calls it with task_func_addr's R0)
116 // *(--stack_ptr) = initial_lparam; // R2 (example, depends on calling convention for 3rd param)
117 // *(--stack_ptr) = initial_sparam; // R1
118 // *(--stack_ptr) = initial_msg_type; // R0
119 // *(--stack_ptr) = 0xCCCCCCCCL; // R12
120 // *(--stack_ptr) = 0xCCCCCCCCL; // R3
121 // *(--stack_ptr) = 0xCCCCCCCCL; // R2 (if not used for lparam)
122 // *(--stack_ptr) = 0xCCCCCCCCL; // R1 (if not used for sparam)
123 // *(--stack_ptr) = 0xCCCCCCCCL; // R0 (if not used for msg_type)
124 // // ... push other registers (R4-R11 often) ...
125 // return stack_ptr;
126
127 (void)p_stack_base;
128 (void)stack_size_bytes;
129 (void)task_func_addr;
130 (void)task_exit_handler_addr;
131 (void)initial_msg_type;
132 (void)initial_sparam;
133 (void)initial_lparam;
134
135 // If stack setup fails (e.g., stack_size_bytes is too small), return NULL.
136 // ktos_Emergency("ktos_hal_InitTaskStack: Not implemented for this BSP!");
137 return NULL;
138}

145void ktos_hal_ContextSwitch(void **p_current_task_sp_storage, void *next_task_sp_val)
146{
147 // TODO: Implement the context switch. This is almost entirely assembly language.
148 //
149 // This function is called with interrupts DISABLED.
150 //
151 // Steps:
152 // 1. Save CPU registers of the current context:
153 // - Push all general-purpose registers (that are not callee-saved by C convention
154 // if switching between C functions, but for OS context switch, save all GPRs).
155 // - Push the Program Status Register (PSR).
156 // - The Stack Pointer (SP) register now points to the top of this saved context.
157 //
158 // 2. Save the current CPU SP (which now points to the saved context on the current stack)
159 // into the location pointed to by `p_current_task_sp_storage`.
160 // Example: `*p_current_task_sp_storage = (void*)__get_PSP();` (ARM Cortex-M Process Stack Pointer)
161 // or `*p_current_task_sp_storage = (void*)__get_MSP();` (ARM Cortex-M Main Stack Pointer)
162 //
163 // 3. Set the CPU SP to `next_task_sp_val`.
164 // Example: `__set_PSP((uint32_t)next_task_sp_val);`
165 // or `__set_MSP((uint32_t)next_task_sp_val);`
166 //
167 // 4. Restore CPU registers for the new context:
168 // - Pop the Program Status Register (PSR) from the new stack.
169 // - Pop all general-purpose registers from the new stack.
170 // - The final operation is usually a special return-from-interrupt/exception instruction
171 // or a branch that loads the PC and PSR, effectively resuming the new context.
172 //
173 // Note on Interrupts:
174 // The ktos_hal_ContextSwitch is called with interrupts disabled.
175 // The act of restoring the PSR for the new task will typically restore its interrupt state.
176 // If the new task was previously running with interrupts enabled, they will become enabled
177 // as part of restoring its PSR.
178
179 (void)p_current_task_sp_storage;
180 (void)next_task_sp_val;
181 // ktos_Emergency("ktos_hal_ContextSwitch: Not implemented for this BSP!");
182}
