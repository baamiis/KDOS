166 void *ktos_hal_InitTaskStack(void *p_stack_base,
167  unsigned int stack_size_bytes,
168  WORD (*task_func_addr)(WORD, WORD, LONG),
169  void (*task_exit_handler_addr)(WORD),
170  WORD initial_msg_type,
171  WORD initial_sparam,
172  LONG initial_lparam)
173 {
174  uint32_t *sp = (uint32_t *)((uint8_t *)p_stack_base + stack_size_bytes);
175  sp = (uint32_t *)((uint32_t)sp & ~0xFUL); /* 16-byte alignment */
176  sp -= 16; /* 16 words = 64-byte frame */
177 
178  sp[0] = (uint32_t)ktos_task_launch; /* a0 — resume PC (trampoline) */
179  sp[1] = 0; /* SAR — zeroed */
180  sp[2] = (uint32_t)initial_msg_type; /* a2 — first task argument */
181  sp[3] = (uint32_t)initial_sparam; /* a3 — second task argument */
182  sp[4] = (uint32_t)initial_lparam; /* a4 — third task argument */
183  sp[5] = (uint32_t)task_func_addr; /* a5 — read by trampoline */
184  sp[6] = (uint32_t)task_exit_handler_addr; /* a6 — read by trampoline */
185  sp[7] = 0; sp[8] = 0; sp[9] = 0; /* a7-a9 */
186  sp[10] = 0; sp[11] = 0; sp[12] = 0; /* a10-a12 */
187  sp[13] = 0; sp[14] = 0; sp[15] = 0; /* a13-a15 */
188 
189  return (void *)sp;
190 }
